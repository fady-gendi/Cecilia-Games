/** Cecilia Automatic Account — v2.6.
 * Use a NEW dedicated project. Existing Apps Script projects stay unchanged.
 * Run setupCeciliaAccount() once, then deploy as a web app: Me / Anyone.
 * No browser family key or per-device account setup is required.
 * This is a keyless endpoint for virtual game data; people who know its URL
 * can read or change Cecilia's game progress and virtual coins.
 * Do not store passwords, contact details or other personal data here.
 */
var CECILIA_PROFILES_=['cecilia','cecilia_sentences','cecilia_match','cecilia_math_quest','cecilia_little_world'];
function setupCeciliaAccount(){
 console.log('1/5 Setup started.');
 var lock=LockService.getScriptLock();
 if(!lock.tryLock(1000))throw Error('Another execution is still running. Stop it in Executions before running setup again.');
 try{
  var p=PropertiesService.getScriptProperties(),id=p.getProperty('WALLET_SHEET_ID'),ss;
  if(id){console.log('2/5 Opening the existing account spreadsheet.');ss=SpreadsheetApp.openById(id)}
  else{
   console.log('2/5 Creating the account spreadsheet. Waiting for Google Sheets...');
   ss=SpreadsheetApp.create('Cecilia Global Account');
   id=ss.getId();p.setProperty('WALLET_SHEET_ID',id);
   console.log('Spreadsheet created and saved.');
  }
  console.log('3/5 Preparing the coin ledger.');
  var ledger=ss.getSheetByName('Ledger')||ss.insertSheet('Ledger');
  if(!ledger.getLastRow())ledger.appendRow(['Id','Source','Kind','Delta','Meta','Created','Status','Error','MigrationTotal','Reason']);
  console.log('4/5 Preparing game saves.');
  var saves=ss.getSheetByName('Game_Saves')||ss.insertSheet('Game_Saves');
  if(!saves.getLastRow())saves.appendRow(['Profile','State','Timestamp']);
  console.log('5/5 READY. Setup completed. Spreadsheet: '+ss.getUrl());
  console.log('Deploy as a web app: execute as Me, access Anyone. No family key needed.');
  return {ok:true,spreadsheetUrl:ss.getUrl()};
 }catch(e){console.error('SETUP FAILED: '+String(e.message||e));throw e}
 finally{lock.releaseLock()}
}
function setupGlobalWallet(){return setupCeciliaAccount()}
function walletBook_(){var id=PropertiesService.getScriptProperties().getProperty('WALLET_SHEET_ID');if(!id)throw Error('Run setupCeciliaAccount first');return SpreadsheetApp.openById(id)}
function json_(o){return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON)}
function doGet(e){try{
 var q=e&&e.parameter||{};
 if(!q.action||q.action==='health')return json_({ok:true,service:'cecilia-auto-account',version:'2.6',account:'cecilia',keyRequired:false});
 if(q.action==='load')return json_(loadGame_(q.p));
 return json_({ok:false,error:'unknown action'});
}catch(x){return json_({ok:false,error:String(x.message||x)})}}
function doPost(e){try{
 var body=e&&e.postData&&e.postData.contents||'';
 if(body.length>500000)return json_({ok:false,error:'request too large'});
 var q=JSON.parse(body);
 if(q.account&&q.account!=='cecilia')return json_({ok:false,error:'bad account'});
 if(q.action==='wallet.sync')return json_(syncWallet_(q));
 if(q.action==='save')return json_(saveGame_(q));
 return json_({ok:false,error:'unknown action'});
}catch(x){return json_({ok:false,error:String(x.message||x)})}}
function syncWallet_(q){
 if(q.account!=='cecilia'||!Array.isArray(q.events)||q.events.length>80)return {ok:false,error:'bad request'};
 var lock=LockService.getScriptLock();lock.waitLock(20000);
 try{
  var sheet=walletBook_().getSheetByName('Ledger'),rows=sheet.getDataRange().getValues().slice(1),byId={},migrations={},balance=0,results=[],added=[],sources=['wg','ss','wm','math','littleworld','parent'];
  rows.forEach(function(r){byId[r[0]]=r;if(r[6]==='accepted'){balance+=Number(r[3])||0;if(r[2]==='migration')migrations[r[1]]=Math.max(migrations[r[1]]||0,Number(r[8])||0)}});
  q.events.forEach(function(e){
   var id=String(e.id||''),src=String(e.source||''),kind=e.kind,delta=Number(e.delta),meta=e.meta||{},reason=String(e.reason||'').slice(0,160);
   if(!/^[A-Za-z0-9:_\-.]{1,160}$/.test(id)||sources.indexOf(src)<0||!Number.isSafeInteger(delta)||Math.abs(delta)>1000000||['migration','earn','spend','parent'].indexOf(kind)<0||JSON.stringify(meta).length>4000||(kind==='spend'&&delta>=0)||(kind!=='spend'&&delta<0)||(kind==='earn'&&delta>200)||(kind==='earn'&&['littleworld','parent'].indexOf(src)>=0)||(kind==='parent'&&src!=='parent')||(kind==='migration'&&['wg','ss','wm','math'].indexOf(src)<0)){
    results.push({id:id,ok:false,error:'invalid event'});return;
   }
   if(kind==='migration'){
    var previous=migrations[src]||0,amount=Math.max(previous,delta),extra=amount-previous;
    if(extra>0){var mid='migration:'+src+':to:'+amount;var mr=[mid,src,kind,extra,'{}',Date.now(),'accepted','',amount,reason];added.push(mr);byId[mid]=mr;balance+=extra;migrations[src]=amount}
    results.push({id:id,ok:true,creditedTotal:amount});return;
   }
   if(byId[id]){var old=byId[id];results.push({id:id,ok:old[6]==='accepted',error:old[7]||''});return}
   var ok=delta>=0||balance+delta>=0,error=ok?'':'insufficient funds';
   var row=[id,src,kind,ok?delta:0,JSON.stringify(meta),Number(e.created)||Date.now(),ok?'accepted':'rejected',error,0,reason];added.push(row);byId[id]=row;if(ok)balance+=delta;
   results.push({id:id,ok:ok,error:error});
  });
  if(added.length)sheet.getRange(sheet.getLastRow()+1,1,added.length,10).setValues(added);
  SpreadsheetApp.flush();
  var grants=rows.concat(added).filter(function(r){return r[2]==='spend'&&r[6]==='accepted'&&r[4]&&r[4]!=='{}'}).map(function(r){return {id:r[0],source:r[1],kind:'spend',delta:Number(r[3]),created:Number(r[5]),status:'confirmed',meta:JSON.parse(r[4])}});
  return {ok:true,account:'cecilia',balance:balance,migrations:migrations,results:results,grants:grants};
 }finally{lock.releaseLock()}
}
function loadGame_(profile){if(CECILIA_PROFILES_.indexOf(profile)<0)return {ok:false,error:'bad profile'};var rows=walletBook_().getSheetByName('Game_Saves').getDataRange().getValues();for(var i=1;i<rows.length;i++)if(rows[i][0]===profile)return {ok:true,state:JSON.parse(rows[i][1])};return {ok:true,state:null}}
function saveGame_(q){
 if(CECILIA_PROFILES_.indexOf(q.p)<0||!q.state||typeof q.state!=='object'||Array.isArray(q.state))return {ok:false,error:'bad state'};
 var text=JSON.stringify(q.state);if(text.length>45000)return {ok:false,error:'save too large'};
 var lock=LockService.getScriptLock();lock.waitLock(20000);
 try{var sheet=walletBook_().getSheetByName('Game_Saves'),rows=sheet.getDataRange().getValues(),at=-1;for(var i=1;i<rows.length;i++)if(rows[i][0]===q.p){at=i+1;break}
  if(at>0&&!q.force){var prior=JSON.parse(rows[at-1][1]);if((Number(prior.total)||0)>(Number(q.state.total)||0)||(Number(prior.ts)||0)>(Number(q.state.ts)||0))return {ok:false,error:'older',state:prior}}
  var data=[[q.p,text,Number(q.state.ts)||Date.now()]];if(at>0)sheet.getRange(at,1,1,3).setValues(data);else sheet.getRange(sheet.getLastRow()+1,1,1,3).setValues(data);return {ok:true,ts:q.state.ts};
 }finally{lock.releaseLock()}
}
